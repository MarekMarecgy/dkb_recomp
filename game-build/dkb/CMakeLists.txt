cmake_minimum_required(VERSION 3.25)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(dkb_module_check CXX)
get_filename_component(DKB_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(DKB_LIB_DIR "${DKB_ROOT}/work/build/dkb-llvm23" CACHE PATH "Compiled DKB module libraries")
set(DKB_CORE_LIB "${DKB_ROOT}/build/runtime-core/nxrt.lib" CACHE FILEPATH "Built headless runtime library")
if(NOT EXISTS "${DKB_CORE_LIB}")
  message(FATAL_ERROR "Build the headless nxrt runtime first, or set DKB_CORE_LIB")
endif()
foreach(name main sdk rtld)
  if(NOT EXISTS "${DKB_LIB_DIR}/dkb_${name}.lib")
    message(FATAL_ERROR "Build the DKB modules first: ${DKB_LIB_DIR}/dkb_${name}.lib")
  endif()
  add_library(dkb_${name} STATIC IMPORTED)
  set_target_properties(dkb_${name} PROPERTIES IMPORTED_LOCATION "${DKB_LIB_DIR}/dkb_${name}.lib")
endforeach()
add_executable(dkb_module_check module_check.cpp)
target_compile_features(dkb_module_check PRIVATE cxx_std_20)
target_compile_definitions(dkb_module_check PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(dkb_module_check PRIVATE "${DKB_ROOT}/runtime/include" "${DKB_ROOT}/recomp/include")
target_link_libraries(dkb_module_check PRIVATE dkb_main dkb_sdk dkb_rtld "${DKB_CORE_LIB}" psapi)
target_link_options(dkb_module_check PRIVATE /HIGHENTROPYVA:NO /STACK:67108864)

add_executable(dkb_boot_probe boot_probe.cpp)
target_compile_features(dkb_boot_probe PRIVATE cxx_std_20)
target_compile_definitions(dkb_boot_probe PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(dkb_boot_probe PRIVATE "${DKB_ROOT}/runtime/include" "${DKB_ROOT}/recomp/include")
target_link_libraries(dkb_boot_probe PRIVATE dkb_main dkb_sdk dkb_rtld "${DKB_CORE_LIB}" psapi)
target_link_options(dkb_boot_probe PRIVATE /HIGHENTROPYVA:NO /STACK:67108864)

# Same probe with the NVN front end and the headless Vulkan back end (needs a graphics build of the runtime).
set(DKB_GFX_DIR "${DKB_ROOT}/build/runtime-gfx" CACHE PATH "Graphics build of the runtime (nxrt_gfx.lib and its dependencies)")
set(DKB_TEXTURES_DIR "${DKB_ROOT}/tools/textures/build/release" CACHE PATH "Built tools/textures libraries")
if(EXISTS "${DKB_GFX_DIR}/nxrt_gfx.lib")
  file(GLOB DKB_SDL_LIB "${DKB_GFX_DIR}/_deps/sdl3-build/SDL3-static.lib")
  add_executable(dkb_boot_probe_gfx boot_probe.cpp)
  target_compile_features(dkb_boot_probe_gfx PRIVATE cxx_std_20)
  target_compile_definitions(dkb_boot_probe_gfx PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN DKB_PROBE_GFX=1)
  target_include_directories(dkb_boot_probe_gfx PRIVATE "${DKB_ROOT}/runtime/include" "${DKB_ROOT}/recomp/include")
  target_include_directories(dkb_boot_probe_gfx SYSTEM PRIVATE "${DKB_GFX_DIR}/_deps/volk-src" "${DKB_GFX_DIR}/_deps/vulkan_headers-src/include"
    "${DKB_GFX_DIR}/_deps/vma-src/include" "${DKB_GFX_DIR}/_deps/imgui-src" "${DKB_GFX_DIR}/_deps/sdl3-src/include" "${DKB_ROOT}/runtime/src/gfx/nvn/third_party")
  target_compile_definitions(dkb_boot_probe_gfx PRIVATE VMA_STATIC_VULKAN_FUNCTIONS=0 VMA_DYNAMIC_VULKAN_FUNCTIONS=1)
  target_link_libraries(dkb_boot_probe_gfx PRIVATE dkb_main dkb_sdk dkb_rtld
    "${DKB_GFX_DIR}/nxrt_gfx.lib" "${DKB_GFX_DIR}/nxrt.lib" "${DKB_GFX_DIR}/nxrt_volk.lib" "${DKB_GFX_DIR}/nxrt_imgui.lib"
    "${DKB_GFX_DIR}/nxrt_tiling.lib" "${DKB_TEXTURES_DIR}/astc2bc7.lib" "${DKB_TEXTURES_DIR}/bc7e_ispc.lib" ${DKB_SDL_LIB}
    psapi user32 gdi32 winmm imm32 ole32 oleaut32 version uuid advapi32 setupapi shell32 cfgmgr32 shlwapi mf mfplat mfreadwrite mfuuid xaudio2 d3d11 dxgi dwmapi)
  target_link_options(dkb_boot_probe_gfx PRIVATE /HIGHENTROPYVA:NO /STACK:67108864 /MAP)
endif()
